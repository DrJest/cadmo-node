1. Guida rapida

Un middleware

function mio(req, res, next) {
  // 1. rispondi:   return res.status(401).json({ … })
  // 2. lascia passare:     next()
  // 3. segnala un errore:  next(err)  (o throw)
}
app.use(mio);                     // tutte le richieste
app.use('/api', mio);             // solo /api/…
app.post('/api/libri', mio, gestore);  // una rotta

File statici

const path = require('path');
app.use(express.static(path.join(__dirname, 'public')));
// GET /           → public/index.html
// GET /style.css  → public/style.css
// file mancante   → next()

app.use('/docs', express.static(path.join(__dirname, 'documenti')));
// GET /docs/a.pdf → documenti/a.pdf

morgan e cors

// npm install morgan cors
const morgan = require('morgan');
const cors = require('cors');

app.use(morgan('dev'));    // GET /api/libri 200 1.2 ms - 182
app.use(cors({ origin: 'http://localhost:5500' }));
// preflight OPTIONS → 204 automatico

Router

// routes/libri.js
const router = express.Router();
router.get('/', elenco);          // GET /api/libri
router.get('/:id', dettaglio);    // GET /api/libri/7
module.exports = router;

// app.js
app.use('/api/libri', require('./routes/libri'));

404 ed errori

// in fondo, dopo tutte le rotte
app.use((req, res) =>
  res.status(404).json({ errore: 'Rotta non trovata' }));

// l'ultimo, con 4 parametri
app.use((err, req, res, next) => {
  const status = err.status || 500;
  if (status === 500) console.error(err);
  res.status(status).json({ errore: status === 500 ? 'Errore interno' : err.message });
});

L'ordine giusto

app.use(morgan('dev'));          // 1. log
app.use(cors({ origin }));       // 2. CORS
app.use(express.json());         // 3. body
app.use(express.static(...));    // 4. file statici
app.use('/api/libri', router);   // 5. rotte e router
app.use(nonTrovato);             // 6. 404
app.use(gestoreErrori);          // 7. errori (4 parametri)

Missione B: Provoca (e risolvi) un errore CORS

Focus: origini, preflight, cors

  1. Nel progetto della missione A commenta la riga di cors e in public/app.js usa l'URL completo http://localhost:3000/api/libri.
  2. Servi la cartella public/ da un'altra origine: npx serve public -l 5500 (oppure l'estensione Live Server di VS Code).
  3. Apri http://localhost:5500: il catalogo resta vuoto. Cosa dice la console? La richiesta compare nel log di morgan? 🤔
  4. Rimetti cors({ origin: [...] }): ora funziona? E da http://127.0.0.1:5500?
  5. Aggiungi nella pagina un pulsante che fa una POST JSON con l'header X-Api-Key: nella tab Network trovi il preflight OPTIONS?
// 3. Console: "Access to fetch at 'http://localhost:3000/api/libri' from origin
//    'http://localhost:5500' has been blocked by CORS policy: No 'Access-Control-Allow-Origin'..."
//    Morgan invece la registra: GET /api/libri 200 … → il server HA risposto,
//    è il BROWSER che nasconde la risposta alla pagina.

// 4. funziona da localhost:5500 e da 127.0.0.1:5500 (sono entrambe nella lista);
//    da un'origine non in lista l'header Access-Control-Allow-Origin manca.
app.use(cors({ origin: ['http://localhost:5500', 'http://127.0.0.1:5500'] }));

// 5. nel browser (public/app.js)
document.querySelector('#aggiungi').addEventListener('click', async () => {
  const r = await fetch('http://localhost:3000/api/libri', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'X-Api-Key': 'cadmo' },
    body: JSON.stringify({ titolo: 'Dune', autore: 'Frank Herbert' }),
  });
  console.log(r.status, await r.json());
});
// Network: prima "OPTIONS /api/libri 204" (preflight), poi "POST /api/libri 201".
// Il preflight c'è per via di Content-Type: application/json e dell'header X-Api-Key.

Missione C: Dal monolite ai moduli

Focus: struttura del progetto, Router

  1. Porta sul PC la soluzione del Lab 8 (boss) con la stessa struttura di cartelle: server.js, app.js, routes/, middleware/, public/.
  2. Aggiungi routes/autori.js con GET / (elenco degli autori, senza doppioni) e GET /:nome/libri, e montalo su /api/autori.
  3. Dentro il nuovo router stampa req.baseUrl, req.url e req.originalUrl: corrispondono a quanto visto nelle slide?
  4. Lancia node server.js da un'altra cartella (es. cd .. && node lezione5-boss/server.js): la home funziona ancora? Perché?
// routes/autori.js  (i libri vanno condivisi: spostali in data/libri.js e importali nei due router)
const express = require('express');
const libri = require('../data/libri');

const router = express.Router();

router.use((req, res, next) => {
  console.log({ baseUrl: req.baseUrl, url: req.url, originalUrl: req.originalUrl });
  next();
});

router.get('/', (req, res) => {
  res.json([...new Set(libri.map((l) => l.autore))]);
});

router.get('/:nome/libri', (req, res) => {
  const nome = req.params.nome.toLowerCase();
  res.json(libri.filter((l) => l.autore.toLowerCase().includes(nome)));
});

module.exports = router;

// app.js:  app.use('/api/autori', require('./routes/autori'));   (prima di nonTrovato!)
// GET /api/autori/calvino/libri → { baseUrl: '/api/autori', url: '/calvino/libri', originalUrl: '/api/autori/calvino/libri' }
// 4. Sì: con path.join(__dirname, 'public') il percorso non dipende dalla cartella corrente.
//    Con express.static('public') invece Express cercherebbe ./public nella cartella da cui lanci node.

Missione D: Log e errori "da produzione"

Focus: opzioni di morgan, NODE_ENV

  1. Fai scrivere a morgan, in formato combined, tutte le richieste nel file access.log (usa fs.createWriteStream con { flags: 'a' } e l'opzione stream).
  2. Nel terminale, invece, mostra con il formato dev solo le risposte con status ≥ 400 (opzione skip).
  3. Segui il file mentre fai richieste: tail -f access.log (su Windows: Get-Content access.log -Wait).
  4. Avvia con NODE_ENV=production e chiama una rotta che fallisce senza il tuo gestore degli errori: cosa cambia nella pagina d'errore di Express rispetto allo sviluppo?
const fs = require('fs');
const path = require('path');
const morgan = require('morgan');

const accessLog = fs.createWriteStream(path.join(__dirname, 'access.log'), { flags: 'a' });

app.use(morgan('combined', { stream: accessLog }));                          // tutto su file
app.use(morgan('dev', { skip: (req, res) => res.statusCode < 400 }));        // solo errori a video

// access.log:
// ::1 - - [18/Oct/2026:10:15:02 +0000] "GET /api/libri HTTP/1.1" 200 182 "-" "curl/8.7.1"

// 4. NODE_ENV=production node server.js
//    sviluppo  → pagina con lo stack dell'errore (file e righe del server!)
//    produzione → solo "Internal Server Error": i dettagli interni non vanno mai mostrati
//    (su Windows PowerShell: $env:NODE_ENV="production"; node server.js)
