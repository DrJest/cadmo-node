3. Biblioteca 6.0

La Biblioteca passa da un file JSON a SQLite: all'avvio db.js applica le migrazioni e il seed, i repository parlano SQL (con le JOIN), i vincoli violati diventano 409. Sotto il log c'è il contenuto di data/biblioteca.db, aggiornato dopo ogni richiesta. Il pulsante ↻ riavvia il server mantenendo il file; quello con il database rimette i dati iniziali.

L'attacco: SQL injection

La stessa ricerca scritta male (/cerca) e bene (/sicuro/cerca). Mettiti nei panni dell'attaccante: il log mostra la query che arriva davvero al database. Scrivi i valori direttamente nell'URL del client HTTP (gli spazi vanno bene).

/cerca?autore=' OR '1'='1
   → WHERE autore = '' OR '1'='1'                         sempre vero: tutti i libri
/cerca?autore=' UNION SELECT nome, email FROM utenti --
   → WHERE autore = '' UNION SELECT nome, email FROM utenti --'
     le righe di un'altra tabella, con lo stesso numero di colonne; -- commenta l'apice finale
/cerca?autore=D'Annunzio
   → WHERE autore = 'D'Annunzio'                          SQL non valido: 500
/sicuro/cerca?autore=' OR '1'='1
   → WHERE autore = ?   con ? = "' OR '1'='1"             nessun autore si chiama così: []

Missione B: I prestiti

Focus: una nuova migrazione, tabella ponte, transazione, JOIN su tre tabelle

  1. Scrivi migrazioni/003-prestiti.sql: una tabella prestiti con libro_id (chiave esterna), lettore (testo obbligatorio), dal e al (date ISO, al vuoto finché il libro non torna). Riavvia: la migrazione parte da sola?
  2. Crea repositories/prestiti.js con presta(libroId, lettore): in una transazione inserisce il prestito e mette il libro non disponibile. Se il libro non c'è o è già in prestito, niente modifiche.
  3. Aggiungi POST /api/libri/:id/prestito ({ "lettore": "Giulia" } → 201, 404 o 409) e DELETE /api/libri/:id/prestito (la restituzione).
  4. Aggiungi GET /api/prestiti: i prestiti in corso con titolo, autore e lettore (JOIN su tre tabelle).
-- migrazioni/003-prestiti.sql
CREATE TABLE prestiti (
  id       INTEGER PRIMARY KEY,
  libro_id INTEGER NOT NULL REFERENCES libri(id),
  lettore  TEXT NOT NULL,
  dal      TEXT NOT NULL DEFAULT (date('now')),
  al       TEXT                                   -- NULL = non ancora restituito
);
CREATE INDEX idx_prestiti_libro ON prestiti(libro_id);
// repositories/prestiti.js
const db = require('../db');

// restituisce 'ok', 'non-trovato' o 'gia-in-prestito'
function presta(libroId, lettore) {
  const libro = db.prepare('SELECT disponibile FROM libri WHERE id = ?').get(libroId);
  if (!libro) return 'non-trovato';
  if (libro.disponibile === 0) return 'gia-in-prestito';
  db.exec('BEGIN');
  try {
    db.prepare('INSERT INTO prestiti (libro_id, lettore) VALUES (?, ?)').run(libroId, lettore);
    db.prepare('UPDATE libri SET disponibile = 0 WHERE id = ?').run(libroId);
    db.exec('COMMIT');
    return 'ok';
  } catch (err) {
    db.exec('ROLLBACK');
    throw err;
  }
}

function restituisci(libroId) {
  db.exec('BEGIN');
  try {
    const { changes } = db.prepare("UPDATE prestiti SET al = date('now') WHERE libro_id = ? AND al IS NULL").run(libroId);
    if (changes) db.prepare('UPDATE libri SET disponibile = 1 WHERE id = ?').run(libroId);
    db.exec('COMMIT');
    return changes > 0;
  } catch (err) {
    db.exec('ROLLBACK');
    throw err;
  }
}

function inCorso() {
  return db.prepare(`
    SELECT p.id, p.dal, p.lettore, l.titolo, a.nome AS autore
    FROM prestiti p
    JOIN libri l  ON l.id = p.libro_id
    JOIN autori a ON a.id = l.autore_id
    WHERE p.al IS NULL
    ORDER BY p.dal`).all();
}

module.exports = { presta, restituisci, inCorso };

// routes/libri.js
router.post('/:id/prestito', valida({ params: idParam, body: z.strictObject({ lettore: z.string().trim().min(1) }) }), (req, res) => {
  const esito = prestiti.presta(req.valido.params.id, req.valido.body.lettore);
  if (esito === 'non-trovato') throw new NonTrovato('Il libro non esiste');
  if (esito === 'gia-in-prestito') return res.status(409).json({ errore: 'Il libro è già in prestito' });
  res.status(201).json(libri.trova(req.valido.params.id));
});
router.delete('/:id/prestito', valida({ params: idParam }), (req, res) => {
  if (!prestiti.restituisci(req.valido.params.id)) throw new NonTrovato('Nessun prestito in corso per questo libro');
  res.json(libri.trova(req.valido.params.id));
});

// app.js
app.get('/api/prestiti', (req, res) => res.json(require('./repositories/prestiti').inCorso()));
