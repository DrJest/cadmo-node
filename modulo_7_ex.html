1. Guida rapida

Uno schema zod

const { z } = require('zod');
const libro = z.object({
  titolo: z.string().trim().min(1, 'Il titolo è obbligatorio'),
  anno: z.int().min(1450).max(2100).optional(),
  genere: z.enum(['romanzo', 'giallo']).optional(),
  disponibile: z.boolean().default(true),
});
const r = libro.safeParse(req.body);
// { success: true, data } oppure { success: false, error }

Gli errori di zod

r.error.issues           // [{ code, path: ['titolo'], message }, …]
z.flattenError(r.error)  // { formErrors, fieldErrors: { titolo: [ … ] } }
z.prettifyError(r.error) // ✖ …  → at titolo
const dettagli = r.error.issues.map((i) =>
  ({ campo: i.path.join('.'), messaggio: i.message }));
// z.strictObject → errore sui campi sconosciuti
// .partial() per la PATCH: ma niente .default() nello schema base!

Query string e params

z.coerce.number().int().min(1).default(1)  // '2' → 2, 'abc' → errore
z.stringbool().optional()                  // 'false' → false
// ❌ z.coerce.boolean(): Boolean('false') === true
// ❌ req.query = r.data  (Express 5: in sola lettura)
router.get('/:id', valida({ params: idParam }), ctrl.dettaglio);
const { id } = req.valido.params;          // già un numero

Classi di errore

class ErroreApi extends Error {
  constructor(status, messaggio, dettagli) {
    super(messaggio);
    this.name = this.constructor.name;
    this.status = status;
    this.dettagli = dettagli;
  }
}
class NonTrovato extends ErroreApi { constructor(m = 'Risorsa non trovata') { super(404, m); } }
throw new NonTrovato(`Il libro ${id} non esiste`);  // Express 5 → gestore

Il gestore centrale

app.use((req, res, next) => next(new NonTrovato('Rotta non trovata')));
app.use((err, req, res, next) => {            // 4 parametri!
  if (err instanceof ErroreApi) return res.status(err.status)
    .json({ errore: err.message, dettagli: err.dettagli });
  if (err.type === 'entity.parse.failed')
    return res.status(400).json({ errore: 'JSON non valido' });
  console.error(err);                                     // log: tutto
  res.status(500).json({ errore: 'Errore interno del server' }); // client: niente
});

I livelli

routes/        URL + middleware (valida, richiedeChiave) → controller          niente logica
controllers/   req.valido → service → status, header e JSON                    niente regole
services/      le regole: throw new NonTrovato(…), new Conflitto(…)            niente req, res, status
repositories/  leggere e scrivere i dati (file JSON, poi SQLite)               niente regole
middleware/errori.js   l'unico posto che trasforma un errore in una risposta
config.js              l'unico che legge process.env

Missione B: express-validator a confronto

Focus: un'altra libreria, stessi principi

  1. Nel progetto della missione A installa express-validator e crea routes/libriEV.js, montato su /api/v2/libri, con una sola rotta POST /.
  2. Valida il body con le catene di body(): le stesse regole di creaLibro (titolo e autore testi non vuoti di massimo 200 caratteri, anno intero facoltativo, genere tra quelli ammessi, disponibile booleano).
  3. Rispondi con lo stesso formato della Biblioteca: 400 { errore, dettagli: [{ campo, messaggio }] }. Usa matchedData(req) per tenere solo i campi validati.
  4. Confronta: quante righe? Che messaggi ricevi se non ne scrivi tu? Che cosa succede a un campo in più? Come riuseresti le regole per la PATCH?
// routes/libriEV.js — npm install express-validator
const express = require('express');
const { body, validationResult, matchedData } = require('express-validator');
const service = require('../services/libriService');
const { GENERI } = require('../schemi/libro');

const router = express.Router();
const testo = (campo) => body(campo).isString().withMessage(`${campo} deve essere un testo`)
  .trim().notEmpty().withMessage(`${campo} non può essere vuoto`)
  .isLength({ max: 200 }).withMessage(`${campo} può avere al massimo 200 caratteri`);

router.post('/',
  testo('titolo'),
  testo('autore'),
  body('anno').optional({ values: 'null' }).isInt({ min: 1450, max: new Date().getFullYear() }).withMessage('anno non valido').toInt(),
  body('genere').optional({ values: 'null' }).isIn(GENERI).withMessage(`genere deve essere uno tra: ${GENERI.join(', ')}`),
  body('disponibile').optional().isBoolean({ strict: true }).withMessage('disponibile deve essere true o false'),
  async (req, res) => {
    const risultato = validationResult(req);
    if (!risultato.isEmpty()) {
      return res.status(400).json({
        errore: 'Dati non validi',
        dettagli: risultato.array().map((e) => ({ campo: e.path, messaggio: e.msg })),
      });
    }
    const libro = await service.crea(matchedData(req));   // solo i campi validati
    res.status(201).location(`${req.baseUrl}/${libro.id}`).json(libro);
  });

module.exports = router;     // in app.js: app.use('/api/v2/libri', require('./routes/libriEV'));

// Osservazioni:
// - senza withMessage il messaggio predefinito è "Invalid value": poco utile;
// - un campo in più ("prezzo") resta in req.body ma non in matchedData: viene ignorato,
//   come con z.object (per rifiutarlo serve un controllo apposta, es. checkExact());
// - "anno": "1965" passa (isInt accetta le stringhe) e toInt lo converte: utile per i form,
//   ma in un'API JSON zod è più severo e chiaro (z.int() vuole un numero);
// - per la PATCH bisogna duplicare le catene con .optional() (oppure usare checkSchema):
//   con zod basta libro.partial().

Missione C: Quando il processo cade

Focus: uncaughtException, unhandledRejection, codici di uscita, riavvii

  1. Crea crash.js con un server Express qualsiasi e un setTimeout che dopo 3 secondi esegue JSON.parse('{ rotto'). Lancialo con node crash.js; echo "codice di uscita: $?". Il gestore degli errori di Express interviene?
  2. Sostituisci il timer con una promise rifiutata e non gestita, per esempio fetch('http://localhost:9999'). Cambia qualcosa?
  3. Aggiungi i gestori process.on('uncaughtException') e process.on('unhandledRejection') che scrivono l'errore (con data e ora) ed escono con codice 1.
  4. Avvialo con node --watch crash.js: dopo il crash il server riparte da solo? E con nodemon? Chi lo farebbe ripartire in produzione?
// crash.js
const express = require('express');
const app = express();
app.get('/', (req, res) => res.send('ok'));
app.use((err, req, res, next) => res.status(500).json({ errore: 'Errore interno del server' }));
app.listen(3000, () => console.log('in ascolto'));

// 1. un errore in un timer: non passa da nessuna richiesta, quindi NON arriva al gestore di Express
setTimeout(() => JSON.parse('{ rotto'), 3000);
//    → SyntaxError stampato e processo terminato: "codice di uscita: 1"

// 2. una promise rifiutata e non gestita: stesso risultato (in Node 15+ termina il processo)
// fetch('http://localhost:9999').then((r) => r.json());
//    → TypeError: fetch failed … [cause]: connect ECONNREFUSED

// 3. l'ultima rete di sicurezza: registrare e uscire (non continuare!)
const ora = () => new Date().toISOString();
process.on('uncaughtException', (err) => {
  console.error(`${ora()} 💥 uncaughtException`, err);
  process.exit(1);
});
process.on('unhandledRejection', (motivo) => {
  console.error(`${ora()} 💥 unhandledRejection`, motivo);
  process.exit(1);
});

// 4. node --watch NON riavvia dopo un crash: aspetta che modifichi un file
//    ("Failed running 'crash.js'. Waiting for file changes before restarting...").
//    Anche nodemon: "[nodemon] app crashed - waiting for file changes before starting...".
//    In produzione lo fa un "process manager": PM2, systemd, oppure Docker con
//    restart: unless-stopped (modulo 17). Il codice di uscita 1 gli dice che qualcosa è andato storto.

Missione D: Un segreto su Git (e come rimediare)

Focus: .gitignore, git rm --cached, la storia di Git, rotazione dei segreti

  1. In una cartella di prova (non nel progetto vero!) fai git init, crea server.js e un .env con API_KEY=segreto-di-prova-123456, poi git add . e git commit. Ops: niente .gitignore.
  2. Ritrova la chiave nella storia con git show HEAD:.env (oppure git log -p).
  3. Rimedia: aggiungi .env al .gitignore, togli il file dal repository senza cancellarlo dal disco e fai un nuovo commit. Ora la chiave è al sicuro?
  4. Che cosa bisognerebbe fare con quella chiave se il repository fosse già su GitHub? Cerca che cosa fa la "push protection" di GitHub.
mkdir prova-segreti && cd prova-segreti && git init
echo "console.log('ciao')" > server.js
echo "API_KEY=segreto-di-prova-123456" > .env
git add . && git commit -m "primo commit"      # 😱 .env è nel commit

git show HEAD:.env                              # API_KEY=segreto-di-prova-123456

# 3. rimedio: smettere di tracciare il file (resta sul disco)
echo ".env" >> .gitignore
git rm --cached .env
git add .gitignore && git commit -m "Non tracciare .env"

git show HEAD:.env       # fatal: path '.env' exists on disk, but not in 'HEAD'
git show HEAD~1:.env     # API_KEY=segreto-di-prova-123456   ← è ancora nella storia!

# 4. la chiave va considerata RUBATA: si genera una chiave nuova e si disattiva la vecchia
#    ("rotazione"). Riscrivere la storia (git filter-repo) non basta: chiunque abbia già
#    clonato o scaricato il repository ha la vecchia copia.
#    GitHub "push protection" blocca il push quando riconosce un segreto noto (chiavi di
#    servizi come AWS, Stripe, GitHub…); il "secret scanning" avvisa per quelli già pubblicati.
#    Abitudine sana: guardare git status prima di ogni git add .
