1. Guida rapida

DataSource

// npm install typeorm better-sqlite3
const { DataSource } = require('typeorm');
const AppDataSource = new DataSource({
  type: 'better-sqlite3',
  database: 'data/biblioteca.db',
  entities: [Autore, Libro],
  synchronize: true,           // solo in sviluppo
  logging: ['query', 'error'], // vedi l'SQL
});
await AppDataSource.initialize();   // prima di listen

Entità

new EntitySchema({
  name: 'Libro', tableName: 'libri',
  columns: {
    id: { type: Number, primary: true, generated: true },
    titolo: { type: String },
    anno: { type: Number, nullable: true },
    isbn: { type: String, nullable: true, unique: true },
    disponibile: { type: Boolean, default: true },
  },
});

Repository

const libri = AppDataSource.getRepository(Libro);
await libri.save(libri.create({ titolo }));  // INSERT
await libri.save(libro);           // con id: UPDATE
await libri.findOneBy({ id });     // o null
await libri.update(id, { anno });  // { affected }
await libri.delete(id);            // { affected }
await libri.count(); await libri.existsBy({ id });

2. Dall'ORM all'SQL e "che cosa succede?"

Dall'ORM all'SQL

Un database in memoria con 5 autori, 7 libri e 5 generi (le entità sono quelle della Biblioteca 7.0). input è una funzione che riceve il repository dei libri (e la DataSource): scegli un esempio o scrivi la tua chiamata, poi guarda nel terminale le query che TypeORM esegue davvero. Ctrl+Invio esegue.

🎯 Cinque sfide per il playground (con le soluzioni)
  1. I libri pubblicati tra il 1950 e il 1970, in ordine di titolo.
  2. Quanti libri non sono disponibili? (un numero, non un array)
  3. I libri di genere memorie, con l'autore.
  4. La seconda pagina da 3 libri in ordine di anno, insieme al totale.
  5. Rendi non disponibili i libri di Calvino con una query, e guarda affected.
// 1
(libri) => libri.find({ where: { anno: Between(1950, 1970) }, order: { titolo: 'ASC' } })
// 2
(libri) => libri.countBy({ disponibile: false })
// 3 (una JOIN per il filtro sulla relazione N-N)
(libri) => libri.find({ where: { generi: { nome: 'memorie' } }, relations: { autore: true } })
// 4
(libri) => libri.findAndCount({ order: { anno: 'ASC' }, skip: 3, take: 3 })
// 5 — update vuole le colonne della tabella: l'id dell'autore, non il nome
(libri) => libri.update({ autore: { id: 1 } }, { disponibile: false })
//      (oppure con il QueryBuilder:)
(libri) => libri.createQueryBuilder().update().set({ disponibile: false }).where('autore_id = :id', { id: 1 }).execute()

3. Biblioteca 7.0

La Biblioteca passa da SQL a mano a TypeORM: quattro entità, due migrazioni eseguite all'avvio, repository con relazioni e QueryBuilder, il prestito in una transazione. Il log mostra ogni query che TypeORM esegue (e le righe di morgan); sotto c'è il contenuto di data/biblioteca.db, aggiornato dopo ogni richiesta. ↻ riavvia il server mantenendo il file; il pulsante con il database rimette i dati iniziali.

5. Missioni sul tuo PC

Missione A: La Biblioteca 7.0 sul tuo PC

Focus: installazione, logging, il problema N+1 dal vivo

  1. Crea corso-node/lezione10, installa express, morgan, zod, typeorm e better-sqlite3, ricopia i file della Biblioteca 7.0 (sezione 3) e avvia con node --watch server.js.
  2. Nel log: quante query fa il primo avvio (migrazioni + seed)? E il secondo? Apri data/biblioteca.db con SQLite Viewer o sqlite3: che cosa c'è nella tabella migrations?
  3. Aggiungi in routes/autori.js la rotta GET /api/autori/lento che carica gli autori con find() e poi, in un ciclo, i libri di ciascuno. Chiamala e conta le righe query:. Poi riscrivila con relations.
  4. In data-source.js prova logging: true, poi logging: ['error']: che cosa cambia? Quale useresti in produzione?

# (se npm install better-sqlite3 fallisce: serve un Node LTS con i binari già pronti, es. 22 o 24)
// routes/autori.js
const { AppDataSource } = require('../data-source');

router.get('/lento', async (req, res) => {
  const autori = await AppDataSource.getRepository('Autore').find();          // 1 query
  for (const a of autori) {                                                    // + 5 query
    a.libri = await AppDataSource.getRepository('Libro').findBy({ autore: { id: a.id } });
  }
  res.json(autori);                    // 6 righe "query:" nel log: con 1000 autori, 1001
});

// la versione giusta: UNA query con la LEFT JOIN
router.get('/veloce', async (req, res) => {
  res.json(await AppDataSource.getRepository('Autore').find({ relations: { libri: true } }));
});

// 4. logging: true  → anche le query "di servizio"; ['error'] → solo le query fallite (con l'errore).
//    In produzione: ['error'] (o un logger su file); 'query' riempie il log e può contenere dati personali.

Missione B: Le migrazioni con la CLI

Focus: migration:generate, run, show, revert; una nuova colonna

  1. Nel progetto della missione A cancella data/biblioteca.db e la cartella migrations, togli migrationsRun e lascia synchronize: false. Genera la prima migrazione dalle entità con migration:generate (-o per JavaScript) e leggila: che cosa ci trovi, oltre ai CREATE TABLE?
  2. Controlla con migration:show, applica con migration:run, ricontrolla.
  3. Aggiungi all'entità Libro la colonna pagine (numero, facoltativa) e genera una seconda migrazione. Applicala, poi annullala con migration:revert e riapplicala.
  4. Rimetti migrationsRun: true: chi applica le migrazioni adesso, e quando? Perché in produzione si preferisce spesso un comando separato (migration:run) prima di avviare il server?
rm -rf data/biblioteca.db migrations

// 1. la migrazione generata (accorciata): i CREATE TABLE con i nomi dei vincoli scelti da TypeORM…
await queryRunner.query(`CREATE TABLE "autori" ("id" integer PRIMARY KEY AUTOINCREMENT NOT NULL, "nome" varchar NOT NULL, CONSTRAINT "UQ_45afd29c77ea3466a884eed3587" UNIQUE ("nome"))`);
await queryRunner.query(`CREATE TABLE "libri" ("id" integer PRIMARY KEY AUTOINCREMENT NOT NULL, "titolo" varchar NOT NULL, … "autore_id" integer NOT NULL, …)`);
await queryRunner.query(`CREATE INDEX "IDX_5283f5b7eda28c32b3fe5ab93b" ON "libri_generi" ("libro_id") `);
// …e poi, per aggiungere le chiavi esterne, il "giro" tipico di SQLite (che non sa fare ALTER TABLE … ADD CONSTRAINT):
await queryRunner.query(`CREATE TABLE "temporary_libri" (… CONSTRAINT "FK_2d56eb2d5b90fba0b5554ef704c" FOREIGN KEY ("autore_id") REFERENCES "autori" ("id") …)`);
await queryRunner.query(`INSERT INTO "temporary_libri"(…) SELECT … FROM "libri"`);
await queryRunner.query(`DROP TABLE "libri"`);
await queryRunner.query(`ALTER TABLE "temporary_libri" RENAME TO "libri"`);
// down() fa tutto al contrario. Con PostgreSQL sarebbero semplici ALTER TABLE … ADD CONSTRAINT.
